1. ARCHITECTURE OVERVIEW
================================================================================
The realistic water system consists of four distinct, complementary layers:

A. CORE 3D WATER BODY (jd):
   - Geometry: Curved parametric tube along a 3D Catmull-Rom spline with UV coordinates.
   - Vertex Shader: Multi-frequency Gerstner wave displacement (longitudinal current
     flow + transverse swells) plus dynamic pointer wake (cursor depression +
     expanding fluid ripple wavelets). Recomputes surface normals dynamically.
   - Fragment Shader:
     * Beer-Lambert absorption: Deep oceanic midnight abyss (#030e28) to rich ocean
       blue (#0760a0) to vibrant Caribbean turquoise (#0cd4f5) to crystal crests (#e0f7ff).
     * Physical Fresnel reflectance (Schlick approximation for water n=1.333).
     * Dancing aquatic caustics: Dual overlapping animated wave interference patterns.
     * Sun specular glints & microfacet sparkles catching directional sunlight.
     * Organic wave crest foam and whitecaps.

B. 3D FLUID STREAMLINES & FOAM CURRENTS (Zd):
   - 140,000 instanced ribbons along the spline with dynamic caustics, wave
     propagation, and fluid wake parting on pointer interaction.

C. 3D SUSPENDED WATER DROPLETS & SPLASH BEADS ($d):
   - 550+ 3D droplets floating and bobbing in the airspace along the water current.
   - Mathematical Ray-Sphere Intersection: Rendered on camera-facing billboard quads
     with analytical spherical normals for 100% smooth spherical curvature, zero polygon
     edges, diamond sun specular glints, bottom caustic glow, and pointer splash explosions.

D. SCREEN-SPACE & LANDING PAGE WATER DROPS (water-drops.js & water-drops.css):
   - Real-world optical lighting model:
     * Dual specular highlights: Sharp pinpoint sun glint at top-left + soft rim fill.
     * Internal refractive caustic glow at bottom edge.
     * Directional contact shadows and ambient occlusion.
     * Subtle backdrop-filter lens refraction and magnification.
   - Natural clustering around Hero title ("Currents that lift & compound"), glass
     navigation pills, and viewport edges.
   - Physics: Surface tension elastic wobble on hover, click splash ripples with
     flying micro-beads, and organic gravity trickling with evaporating wet trails.

/**
 * Photorealistic Interactive Water Droplets & Liquid Condensation System

(function () {
  'use strict';

  function initWaterDroplets() {
    const container = document.createElement('div');
    container.className = 'water-droplets-container';
    container.id = 'water-droplets-root';
    document.body.appendChild(container);

    const shapes = ['shape-organic-1', 'shape-organic-2', 'shape-organic-3', 'shape-elongated', 'shape-bead'];
    const sizes = ['size-micro', 'size-small', 'size-medium', 'size-large'];

    function createDrop(x, y, sizeClass, shapeClass, options = {}) {
      const drop = document.createElement('div');
      drop.className = `water-drop ${sizeClass || 'size-small'} ${shapeClass || 'shape-organic-1'}`;
      drop.style.left = `${x}px`;
      drop.style.top = `${y}px`;

      if (options.rotation) {
        drop.style.transform = `rotate(${options.rotation}deg)`;
      }

      drop.addEventListener('mouseenter', () => {
        triggerWobble(drop);
      });

      drop.addEventListener('click', (e) => {
        e.stopPropagation();
        triggerSplash(e.clientX, e.clientY);
        triggerWobble(drop);
      });

      container.appendChild(drop);
      return drop;
    }

    function triggerWobble(drop) {
      drop.classList.remove('wobble');
      void drop.offsetWidth;
      drop.classList.add('wobble');
    }

    function triggerSplash(x, y) {
      const ripple = document.createElement('div');
      ripple.className = 'water-ripple';
      ripple.style.left = `${x}px`;
      ripple.style.top = `${y}px`;
      ripple.style.width = '50px';
      ripple.style.height = '50px';
      container.appendChild(ripple);
      setTimeout(() => ripple.remove(), 800);

      const beadCount = 6;
      for (let i = 0; i < beadCount; i++) {
        const bead = document.createElement('div');
        bead.className = 'water-splash-bead';
        const size = Math.random() * 4 + 2;
        bead.style.width = `${size}px`;
        bead.style.height = `${size}px`;
        bead.style.left = `${x}px`;
        bead.style.top = `${y}px`;

        const angle = (i / beadCount) * Math.PI * 2 + (Math.random() - 0.5) * 0.5;
        const dist = Math.random() * 32 + 18;
        const dx = Math.cos(angle) * dist;
        const dy = Math.sin(angle) * dist;

        bead.style.setProperty('--dx', `${dx}px`);
        bead.style.setProperty('--dy', `${dy}px`);
        container.appendChild(bead);
        setTimeout(() => bead.remove(), 600);
      }
    }

    function spawnCluster(centerX, centerY, radiusX, radiusY, count) {
      for (let i = 0; i < count; i++) {
        const u = Math.random();
        const v = Math.random();
        const rX = (u - 0.5) * 2 * radiusX;
        const rY = (v - 0.5) * 2 * radiusY;
        const x = Math.max(10, Math.min(window.innerWidth - 30, centerX + rX));
        const y = Math.max(10, Math.min(window.innerHeight - 30, centerY + rY));

        let size = 'size-micro';
        const rand = Math.random();
        if (rand > 0.85) size = 'size-large';
        else if (rand > 0.6) size = 'size-medium';
        else if (rand > 0.3) size = 'size-small';

        const shape = shapes[Math.floor(Math.random() * shapes.length)];
        const rot = Math.floor(Math.random() * 360);
        createDrop(x, y, size, shape, { rotation: rot });
      }
    }

    function layoutDroplets() {
      container.innerHTML = '';
      const w = window.innerWidth;
      const h = window.innerHeight;

      // 1. Cluster around Hero Heading
      const heroHeading = document.querySelector('.hero-heading');
      if (heroHeading) {
        const rect = heroHeading.getBoundingClientRect();
        spawnCluster(rect.left + rect.width * 0.22, rect.top + rect.height * 0.45, 90, 45, 14);
        spawnCluster(rect.left + rect.width * 0.78, rect.top + rect.height * 0.65, 80, 40, 12);
        spawnCluster(rect.left + rect.width * 0.50, rect.top + rect.height * 0.15, 120, 25, 10);
      } else {
        spawnCluster(w * 0.5, h * 0.32, 260, 90, 30);
      }

      // 2. Dew beads on the Navigation Bar glass pills
      const navPills = document.querySelector('.nav-pills');
      if (navPills) {
        const rect = navPills.getBoundingClientRect();
        createDrop(rect.left + 8, rect.top - 3, 'size-small', 'shape-bead');
        createDrop(rect.left + 24, rect.bottom - 4, 'size-micro', 'shape-bead');
        createDrop(rect.right - 18, rect.top - 2, 'size-small', 'shape-organic-1');
        createDrop(rect.right - 38, rect.bottom - 3, 'size-micro', 'shape-bead');
      }

      // 3. Dew drops resting around the Hero CTA button
      const heroCta = document.querySelector('.hero-cta');
      if (heroCta) {
        const rect = heroCta.getBoundingClientRect();
        createDrop(rect.right + 12, rect.top + 8, 'size-medium', 'shape-organic-2');
        createDrop(rect.right + 26, rect.top + 22, 'size-micro', 'shape-bead');
        createDrop(rect.left - 14, rect.bottom - 6, 'size-small', 'shape-organic-3');
      }

      // 4. Natural perimeter condensation beads
      spawnCluster(w * 0.08, h * 0.12, 60, 50, 8);
      spawnCluster(w * 0.92, h * 0.14, 70, 55, 9);
      spawnCluster(w * 0.04, h * 0.52, 40, 90, 10);
      spawnCluster(w * 0.95, h * 0.58, 40, 110, 12);
      spawnCluster(w * 0.15, h * 0.88, 80, 45, 10);
      spawnCluster(w * 0.82, h * 0.86, 90, 50, 11);

      // 5. Ambient floating mist particles
      for (let i = 0; i < 14; i++) {
        const mist = document.createElement('div');
        mist.className = 'ambient-mist-particle';
        const mSize = Math.random() * 8 + 4;
        mist.style.width = `${mSize}px`;
        mist.style.height = `${mSize}px`;
        mist.style.left = `${Math.random() * w}px`;
        mist.style.top = `${Math.random() * h}px`;
        mist.style.animationDelay = `${(Math.random() * 6).toFixed(1)}s`;
        mist.style.animationDuration = `${(Math.random() * 5 + 6).toFixed(1)}s`;
        container.appendChild(mist);
      }
    }

    // Spawn water drop on clicking empty space
    document.addEventListener('click', (e) => {
      if (e.target.closest('a, button, input, [role="dialog"], .guide-sheet')) return;
      triggerSplash(e.clientX, e.clientY);
      const size = Math.random() > 0.5 ? 'size-small' : 'size-medium';
      const shape = shapes[Math.floor(Math.random() * shapes.length)];
      const drop = createDrop(e.clientX - 8, e.clientY - 8, size, shape);
      drop.style.transform = 'scale(0)';
      setTimeout(() => {
        drop.style.transform = 'scale(1)';
        triggerWobble(drop);
      }, 10);
    });

    // Gentle gravity trickling
    function scheduleTrickle() {
      const drops = container.querySelectorAll('.water-drop.size-large, .water-drop.size-medium');
      if (drops.length > 0) {
        const randomDrop = drops[Math.floor(Math.random() * drops.length)];
        if (randomDrop && !randomDrop.classList.contains('trickling')) {
          randomDrop.classList.add('trickling');
          const currentTop = parseFloat(randomDrop.style.top) || 100;
          const currentLeft = parseFloat(randomDrop.style.left) || 100;
          const distance = Math.floor(Math.random() * 24 + 12);

          const trail = document.createElement('div');
          trail.className = 'water-trail';
          trail.style.left = `${currentLeft + 7}px`;
          trail.style.top = `${currentTop + 4}px`;
          trail.style.height = `${distance}px`;
          container.appendChild(trail);
          setTimeout(() => {
            trail.style.opacity = '0';
            setTimeout(() => trail.remove(), 1600);
          }, 600);

          randomDrop.style.top = `${currentTop + distance}px`;
          randomDrop.classList.add('shape-elongated');

          setTimeout(() => {
            randomDrop.classList.remove('trickling');
            triggerWobble(randomDrop);
          }, 1800);
        }
      }
      setTimeout(scheduleTrickle, Math.random() * 5000 + 4000);
    }

    layoutDroplets();
    setTimeout(scheduleTrickle, 3000);

    let resizeTimer;
    window.addEventListener('resize', () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(layoutDroplets, 250);
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initWaterDroplets);
  } else {
    initWaterDroplets();
  }
})();

function Yd({attack:i=12,release:e=5.5,follow:t=9.2,snapBelow:n=.02}={}){
  let r=0,s=0;
  const o={x:9999,y:9999,z:9999};
  return{
    update(a,l){
      const c=r<n,u=l?1:0,d=l?i:e;
      if(s+=(-d*d*(r-u)-2*d*s)*a,r+=s*a,r>=1&&(r=1,s=0),(r<=0||!l&&r<.001)&&(r=0,s=0),l)
        if(c)o.x=l.x,o.y=l.y,o.z=l.z;
        else{const f=1-Math.exp(-t*a);o.x+=(l.x-o.x)*f,o.y+=(l.y-o.y)*f,o.z+=(l.z-o.z)*f}
      return{strength:r,x:o.x,y:o.y,z:o.z}
    }
  }

function jd(i, e, t, n, r) {
  const a = [], l = [], uvs = [], c = [];
  for (let f = 0; f <= 220; f++) {
    const uCoord = f / 220;
    const { P: m, B: g, N: _ } = r(uCoord);
    for (let p = 0; p <= 28; p++) {
      const vCoord = p / 28;
      const h = vCoord * Math.PI * 2, A = Math.cos(h), S = Math.sin(h);
      a.push(m.x + _.x * t * A + g.x * n * S, m.y + _.y * t * A + g.y * n * S, m.z + _.z * t * A + g.z * n * S);
      const T = _.x * A / t + g.x * S / n, C = _.y * A / t + g.y * S / n, w = _.z * A / t + g.z * n * S, b = Math.hypot(T, C, w);
      l.push(T / b, C / b, w / b);
      uvs.push(uCoord, vCoord);
    }
  }
  for (let f = 0; f < 220; f++)
    for (let m = 0; m < 28; m++) {
      const g = f * 29 + m, _ = g + 28 + 1;
      c.push(g, _, g + 1, _, _ + 1, g + 1);
    }
  const u = new Vt;
  u.setAttribute("position", new Et(a, 3));
  u.setAttribute("normal", new Et(l, 3));
  u.setAttribute("uv", new Et(uvs, 2));
  u.setIndex(c);

  const waterUniforms = {
    uTime: { value: 0 },
    uMouse: { value: new P(9999, 9999, 9999) },
    uPush: { value: 0 }
  };

  const waterMat = new un({
    uniforms: waterUniforms,
    side: Lt,
    transparent: true,
    depthWrite: true,
    vertexShader: `
      uniform float uTime;
      uniform vec3 uMouse;
      uniform float uPush;
      varying vec2 vUv;
      varying vec3 vNormal;
      varying vec3 vWorldPos;
      varying vec3 vViewPos;
      varying float vWave;

      void main() {
        vUv = uv;
        vec3 pos = position;
        vec3 nrm = normalize(normal);

        // Multi-frequency liquid flow along current (uv.x) + transverse swells (uv.y)
        float flow = uv.x * 24.0 - uTime * 3.4;
        float swirl = uv.y * 10.0 + uTime * 1.8;

        float w1 = sin(flow) * cos(swirl * 0.8) * 0.07;
        float w2 = sin(flow * 2.4 + swirl * 1.3) * 0.038;
        float w3 = cos(flow * 4.8 - uTime * 3.0) * sin(swirl * 2.2) * 0.02;
        float w4 = sin(pos.x * 2.8 + pos.z * 3.8 - uTime * 4.2) * 0.025;
        float totalWave = w1 + w2 + w3 + w4;

        // Pointer wake & radial fluid depression
        vec3 toM = pos - uMouse;
        float dist = length(toM);
        float wake = smoothstep(2.6, 0.0, dist) * uPush;
        totalWave += sin(dist * 14.0 - uTime * 7.5) * wake * 0.14;
        totalWave -= wake * 0.09;

        pos += nrm * totalWave;
        vWave = totalWave;

        // Dynamic normal perturbation for glistening wave facets
        float dU = cos(flow) * 0.07 * 24.0 + cos(flow * 2.4) * 0.038 * 2.4;
        vec3 perturbedN = normalize(nrm + vec3(dU * 0.3, totalWave * 0.6, -dU * 0.25));

        vNormal = normalize(normalMatrix * perturbedN);
        vec4 worldP = modelMatrix * vec4(pos, 1.0);
        vWorldPos = worldP.xyz;
        vec4 mvPos = modelViewMatrix * vec4(pos, 1.0);
        vViewPos = -mvPos.xyz;
        gl_Position = projectionMatrix * mvPos;
      }
    `,
    fragmentShader: `
      precision highp float;
      uniform float uTime;
      varying vec2 vUv;
      varying vec3 vNormal;
      varying vec3 vWorldPos;
      varying vec3 vViewPos;
      varying float vWave;

      void main() {
        vec3 N = normalize(vNormal);
        vec3 V = normalize(vViewPos);
        vec3 L = normalize(vec3(-0.45, 0.85, 0.4));
        vec3 H = normalize(L + V);

        // 1. Deep Ocean Absorption & Subsurface Scattering Colors
        vec3 deepAbyss = vec3(0.012, 0.045, 0.16);
        vec3 oceanMid  = vec3(0.028, 0.38, 0.72);
        vec3 cyanShallow = vec3(0.08, 0.78, 0.98);
        vec3 crestGlint = vec3(0.88, 0.97, 1.0);

        // Fresnel reflection (Schlick approximation)
        float NdotV = max(dot(N, V), 0.0);
        float fresnel = 0.035 + 0.965 * pow(1.0 - NdotV, 4.0);

        // Depth modulation from wave displacement
        float depthF = clamp((vWave + 0.08) / 0.16, 0.0, 1.0);
        vec3 waterCol = mix(deepAbyss, oceanMid, smoothstep(0.04, 0.52, depthF));
        waterCol = mix(waterCol, cyanShallow, smoothstep(0.42, 0.88, depthF));

        // 2. High-Fidelity Dancing Aquatic Caustics
        vec2 cUv1 = vUv * vec2(28.0, 14.0) + vec2(-uTime * 0.55, uTime * 0.25);
        vec2 cUv2 = vUv * vec2(22.0, 18.0) + vec2(uTime * 0.45, -uTime * 0.35);
        float c1 = sin(cUv1.x + cos(cUv1.y * 1.5)) * sin(cUv1.y + sin(cUv1.x * 1.2));
        float c2 = cos(cUv2.x + sin(cUv2.y * 1.3)) * cos(cUv2.y + cos(cUv2.x * 1.1));
        float caustic = pow(clamp((c1 + c2) * 0.5 + 0.5, 0.0, 1.0), 2.8);
        waterCol += cyanShallow * caustic * 0.42;

        // 3. Sun Specular Highlights (Microfacet Glints)
        float NdotH = max(dot(N, H), 0.0);
        float spec = pow(NdotH, 52.0);
        float sparkle = pow(NdotH, 180.0) * (0.65 + 0.35 * sin(uTime * 14.0 + vWorldPos.x * 22.0 + vWorldPos.z * 16.0));
        vec3 sunGlint = crestGlint * (spec * 1.3 + sparkle * 2.6);

        // 4. Foam / Aeration on Wave Crests
        float foamNoise = sin(vUv.x * 85.0 - uTime * 6.5) * cos(vUv.y * 45.0 + uTime * 4.5);
        float foam = smoothstep(0.05, 0.10, vWave + foamNoise * 0.025);
        vec3 foamCol = crestGlint * foam * 0.75;

        // Composite
        vec3 finalCol = mix(waterCol, crestGlint * 0.82, fresnel * 0.65);
        finalCol += sunGlint + foamCol;

        float alpha = clamp(0.88 + fresnel * 0.12, 0.85, 0.98);
        gl_FragColor = vec4(finalCol, alpha);
        #include <tonemapping_fragment>
        #include <colorspace_fragment>
      }
    `
  });

  const d = new Dt(u, waterMat);
  i.add(d);
  return { mesh: d, uniforms: waterUniforms };
}

function Zd(i, e, t, n, r, s) {
  const l = new Wd, c = [], u = [], d = [];
  for (let S = 0; S <= 3; S++) {
    const T = S / 3, C = 0.5 * (1 - T * T);
    c.push(-C, T, 0, C, T, 0), u.push(0, T, 1, T);
  }
  c[c.length - 6] = 0, c[c.length - 3] = 0;
  for (let S = 0; S < 3; S++) {
    const T = S * 2, C = T + 1, w = T + 2, b = T + 3;
    d.push(T, C, w, C, b, w);
  }
  l.setAttribute("position", new Et(c, 3)), l.setAttribute("uv", new Et(u, 2)), l.setIndex(d);
  const f = new Float32Array(14e4 * 3), m = new Float32Array(14e4 * 3), g = new Float32Array(14e4 * 4);
  let _ = 0;
  for (; _ < 14e4;) {
    const S = Math.random(), T = Math.random() * Math.PI * 2, C = Math.cos(T), w = Math.sin(T), b = Math.max(C, 0);
    if (Math.random() > 0.3 + 0.7 * b) continue;
    const { P: k, B: x, N: y } = r(S);
    f[_ * 3] = k.x + y.x * t * C + x.x * n * w;
    f[_ * 3 + 1] = k.y + y.y * t * C + x.y * n * w;
    f[_ * 3 + 2] = k.z + y.z * t * C + x.z * n * w;
    let z = y.x * C / t + x.x * w / n, G = y.y * C / t + x.y * w / n, Z = y.z * C / t + x.z * w / n;
    const L = Math.hypot(z, G, Z);
    m[_ * 3] = z / L, m[_ * 3 + 1] = G / L, m[_ * 3 + 2] = Z / L;
    g[_ * 4] = Math.random() * Math.PI * 2;
    g[_ * 4 + 1] = (0.18 + 0.5 * b * b) * (0.6 + 0.8 * Math.random());
    g[_ * 4 + 2] = (Math.random() - 0.5) * 0.7;
    g[_ * 4 + 3] = Math.random();
    _++;
  }
  l.setAttribute("offset", new ci(f, 3)), l.setAttribute("nrm", new ci(m, 3)), l.setAttribute("rand", new ci(g, 4)), l.instanceCount = 14e4;
  const p = {
    uTime: { value: 0 },
    uMouse: { value: new P(9999, 9999, 9999) },
    uMouseR: { value: 2 },
    uPush: { value: 0 }
  };
  const h = new un({
    uniforms: p,
    side: Lt,
    vertexShader: `
      attribute vec3 offset;
      attribute vec3 nrm;
      attribute vec4 rand;
      uniform float uTime;
      uniform vec3 uMouse;
      uniform float uMouseR;
      uniform float uPush;
      varying float vT;
      varying float vShade;
      varying float vDark;
      varying vec3 vWorld;
      varying vec4 vRand;

      void main() {
        float t = uv.y;
        vT = t;
        vRand = rand;
        float len = rand.y;

        vec3 ref = abs(nrm.y) < 0.95 ? vec3(0.0, 1.0, 0.0) : vec3(1.0, 0.0, 0.0);
        vec3 T0 = normalize(cross(nrm, ref));
        vec3 B0 = cross(nrm, T0);
        float ca = cos(rand.x), sa = sin(rand.x);
        vec3 widthDir = T0 * ca + B0 * sa;
        vec3 leanDir  = T0 * -sa + B0 * ca;

        float bend = t * t;

        float wave1 = sin(uTime * 2.8 + offset.x * 1.5 + offset.y * 1.2 + rand.x * 2.0) * 0.16;
        float wave2 = cos(uTime * 2.0 + offset.x * 0.8 + offset.z * 1.4) * 0.10;
        float wave3 = sin(uTime * 3.8 + offset.x * 2.4 + rand.z * 3.0) * 0.06;
        float wave = wave1 + wave2 + wave3;

        vec3 world = offset
                   + nrm * (t * len * 0.9 + sin(uTime * 3.2 + rand.x * 4.0) * 0.03 * t)
                   + widthDir * (position.x * 0.036)
                   + leanDir * (rand.z * 0.28 * len) * bend
                   + (T0 * wave + B0 * wave * 0.75) * bend * len * 2.6;

        vec3 toB = offset - uMouse;
        float d = length(toB);
        float infl = smoothstep(uMouseR, 0.0, d);
        infl *= infl;
        vec3 pushDir = toB - nrm * dot(toB, nrm);
        float pl = length(pushDir);
        pushDir = pl > 0.0001 ? pushDir / pl : T0;
        infl *= uPush;

        world += pushDir * infl * bend * (0.42 + len * 2.8);
        world -= nrm * infl * bend * len * 0.95;
        world += nrm * sin(d * 14.0 - uTime * 6.5) * infl * 0.10 * bend;

        vDark = infl;
        vWorld = world;

        vShade = (0.75 + 0.25 * rand.w) * (0.85 + 0.15 * sin(rand.x * 2.0));
        vShade *= 0.6 + 0.4 * clamp(nrm.y * 0.5 + 0.6, 0.0, 1.0);

        gl_Position = projectionMatrix * modelViewMatrix * vec4(world, 1.0);
      }
    `,
    fragmentShader: `
      precision highp float;
      uniform float uTime;
      varying float vT;
      varying float vShade;
      varying float vDark;
      varying vec3 vWorld;
      varying vec4 vRand;

      void main() {
        vec3 deepSea   = vec3(0.015, 0.06, 0.20);
        vec3 oceanMid  = vec3(0.03, 0.42, 0.78);
        vec3 cyanFlow  = vec3(0.08, 0.82, 0.98);
        vec3 foamGlint = vec3(0.92, 0.98, 1.00);

        float t = vT;
        vec3 col = mix(deepSea, oceanMid, smoothstep(0.0, 0.52, t));
        col = mix(col, cyanFlow, smoothstep(0.42, 0.86, t));

        col += cyanFlow * (vDark * 0.65);

        col *= vShade;
        gl_FragColor = vec4(col, 1.0);
        #include <tonemapping_fragment>
        #include <colorspace_fragment>
      }
    `
  });
  const A = new Dt(l, h);
  A.frustumCulled = false;
  i.add(A);
  return { uniforms: p };
}

function $d(scene, camera, curve, radX, radY, frameFn, upVec) {
  const DROPLET_COUNT = 550;
  const geom = new Wd();
  const pos = [-0.5, -0.5, 0,  0.5, -0.5, 0,  0.5, 0.5, 0,  -0.5, 0.5, 0];

  geom.setAttribute("uv", new Et(uvs, 2));
  geom.setIndex(idx);

  const offsets = new Float32Array(DROPLET_COUNT * 3);
  const initialData = new Float32Array(DROPLET_COUNT * 4);

  for (let i = 0; i < DROPLET_COUNT; i++) {
    const t = Math.random();
    const angle = Math.random() * Math.PI * 2;
    const radial = 1.08 + Math.random() * 0.85;
    const scale = 0.07 + Math.random() * 0.14;

    const { P: pt, B: binorm, N: norm } = frameFn(t);
    const cosA = Math.cos(angle), sinA = Math.sin(angle);
    offsets[i * 3]     = pt.x + norm.x * radX * cosA * radial + binorm.x * radY * sinA * radial;
    offsets[i * 3 + 1] = pt.y + norm.y * radX * cosA * radial + binorm.y * radY * sinA * radial;
    offsets[i * 3 + 2] = pt.z + norm.z * radX * cosA * radial + binorm.z * radY * sinA * radial;

    initialData[i * 4]     = t;
    initialData[i * 4 + 1] = angle;
    initialData[i * 4 + 2] = radial;
    initialData[i * 4 + 3] = scale;
  }

  geom.setAttribute("offset", new ci(offsets, 3));
  geom.setAttribute("aData", new ci(initialData, 4));
  geom.instanceCount = DROPLET_COUNT;

  const dropletUniforms = {
    uTime: { value: 0 },
    uMouse: { value: new P(9999, 9999, 9999) },
    uPush: { value: 0 }
  };

  const dropletMat = new un({
    uniforms: dropletUniforms,
    transparent: true,
    depthWrite: false,
    side: Lt,
    vertexShader: `
      attribute vec3 offset;
      attribute vec4 aData;
      uniform float uTime;
      uniform vec3 uMouse;
      uniform float uPush;
      varying vec2 vUv;
      varying float vScale;

        vUv = uv;

        worldPos.y += sin(uTime * 2.8 + aData.x * 12.0) * 0.05;

          worldPos += normalize(toM + vec3(0.01)) * force * 0.75;
          worldPos.y += force * 0.35;
        }

        vec4 mvPosition = modelViewMatrix * vec4(worldPos, 1.0);
        mvPosition.xy += position.xy * scale;
        gl_Position = projectionMatrix * mvPosition;
      }
    `,
    fragmentShader: `
      precision highp float;
      varying vec2 vUv;
      varying float vScale;

      void main() {
        vec2 coord = vUv * 2.0 - 1.0;
        float r2 = dot(coord, coord);
        if (r2 > 1.0) discard;

        float z = sqrt(1.0 - r2);
        vec3 N = normalize(vec3(coord, z));
        vec3 sunDir = normalize(vec3(-0.45, 0.85, 0.4));
        vec3 V = vec3(0.0, 0.0, 1.0);
        vec3 H = normalize(sunDir + V);

        float NdotH = max(dot(N, H), 0.0);
        float spec = pow(NdotH, 68.0);
        float sparkle = pow(NdotH, 180.0) * 1.6;

        float caustic = pow(clamp(-N.y * 0.7 - N.x * 0.3 + 0.35, 0.0, 1.0), 2.5);
        float fresnel = pow(1.0 - N.z, 2.8);

        vec3 deepWater = vec3(0.03, 0.42, 0.80);
        vec3 cyanCaustic = vec3(0.15, 0.85, 1.0);
        vec3 sunWhite = vec3(1.0, 1.0, 1.0);

        col += cyanCaustic * fresnel * 0.75;

        float alpha = clamp(0.70 + fresnel * 0.28 + spec * 0.85, 0.0, 0.98);
        gl_FragColor = vec4(col, alpha);
        #include <tonemapping_fragment>
        #include <colorspace_fragment>
      }
    `
  });

  const dropletMesh = new Dt(geom, dropletMat);
  dropletMesh.frustumCulled = false;
  scene.add(dropletMesh);

  return {
    uniforms: dropletUniforms,
    update: (delta) => {
      dropletUniforms.uTime.value += delta;
    }
  };
}

function Send-Response {
    param($Response, [int]$StatusCode, [string]$ContentType, [byte[]]$ContentBytes)

        $Response.ContentType = $ContentType
        $Response.AddHeader("Access-Control-Allow-Origin", "*")

        } elseif ($decodedUrl -eq "/radar") {
            $targetFile = Join-Path $rootDir "pages\radar.html"
        } elseif ($decodedUrl -like "/trade*") {
            $targetFile = Join-Path $rootDir "pages\trade_eth.html"
        } elseif ($decodedUrl -eq "/portfolio") {
            $targetFile = Join-Path $rootDir "pages\portfolio.html"
        } elseif ($decodedUrl -eq "/yield") {
            $targetFile = Join-Path $rootDir "pages\yield.html"
        } elseif ($decodedUrl -eq "/theses") {
            $targetFile = Join-Path $rootDir "pages\theses.html"
        } elseif ($decodedUrl -eq "/recharge") {
            $targetFile = Join-Path $rootDir "pages\recharge.html"
        } elseif ($decodedUrl -eq "/launch") {
            $targetFile = Join-Path $rootDir "pages\launch.html"
        } elseif ($decodedUrl -eq "/docs") {
            $targetFile = Join-Path $rootDir "pages\docs.html"
        } elseif ($decodedUrl -like "/docs/*") {
            $slug = $decodedUrl.Substring(6).Trim("/")
            $docPath = Join-Path $rootDir "pages\docs\$slug.html"
            $targetFile = if (Test-Path $docPath) { $docPath } else { Join-Path $rootDir "pages\docs.html" }
        } else {
            $relPath = $decodedUrl.TrimStart("/").Replace("/", "\")
            $candidates = @((Join-Path $rootDir $relPath), (Join-Path $rootDir "public\$relPath"), (Join-Path $rootDir "pages\$relPath"))
            foreach ($c in $candidates) {
                if (Test-Path -Path $c -PathType Leaf) { $targetFile = $c; break }
            }
        }

        if ($targetFile -and (Test-Path -Path $targetFile -PathType Leaf)) {
            $ext = [System.IO.Path]::GetExtension($targetFile).ToLower()
            $mime = if ($mimeTypes.ContainsKey($ext)) { $mimeTypes[$ext] } else { "application/octet-stream" }
            $fileBytes = [System.IO.File]::ReadAllBytes($targetFile)
            Send-Response -Response $response -StatusCode 200 -ContentType $mime -ContentBytes $fileBytes
        } else {
            $notFound = [System.Text.Encoding]::UTF8.GetBytes("<h1>404 Not Found</h1>")
            Send-Response -Response $response -StatusCode 404 -ContentType "text/html; charset=utf-8" -ContentBytes $notFound
        }
    } catch {}
}
